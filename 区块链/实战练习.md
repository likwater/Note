pragma solidity ^0.8.17;

contract Coin{
    address hoster;
    mapping(address=>uint) private balances; //防止非法访问
    event Deal(address from, address to, uint amount);
    event Send(address to, uint amount);
    constructor() {
        hoster = msg.sender;
    }

    function send(address receiver, uint account) public{
        require(msg.sender == hoster);
        balances[receiver] = balances[receiver] + account;
        emit Send(receiver,account);
    }

    function deal(address receiver, uint account) public{
        require(balances[msg.sender]>=account && balances[receiver] + account <= type(uint).max); //防止溢出
        balances[receiver] += account;
        balances[msg.sender] -= account;
        emit Deal(msg.sender, receiver, account);  
    }
        
    function get(address site) public view returns(uint){
        require((msg.sender == hoster) || (site == msg.sender));
        return balances[site];
    } 
}

# 简单投票合约
1. 智能合约

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

// 定义一个名为 Ballot 的投票合约
contract Ballot {
    // 定义投票者结构体
    struct Voter {
        uint8 weight;      // 投票者的权重
        bool voted;        // 投票者是否已投票
        uint8 vote;        // 投票者投的提案索引
        address delegate;  // 投票者委托的地址
    }
    
    address public chairperson; // 主持人的地址
    uint8 number_proposals;     // 提案数量
    mapping(address => Voter) voters; // 存储每个投票者的信息
    uint8[] proposals;          // 存储每个提案的投票数

    // 构造函数，初始化合约
    constructor (uint8 number) {
        chairperson = msg.sender; // 设置合约创建者为主持人
        voters[msg.sender].weight = 1;  // 主持人权重为 1
        number_proposals = number;  // 设置提案数量

        // 初始化提案的投票数为 0
        for (uint i = 0; i < number_proposals; i++)    
            proposals.push(0);
    }

    // 给指定地址的投票者赋予投票权
    function giveRightToVote(address voter) public {
        require(!voters[voter].voted, "you has voted"); // 确保该投票者尚未投票
        require(msg.sender == chairperson, "you don't have the right"); // 只有主持人可以执行此操作
        voters[voter].weight = 1; // 设置投票者的权重为 1
    }

    // 委托投票给其他地址
    function delegate(address to) public {
        Voter storage sender = voters[msg.sender]; // 获取当前发送者的投票者信息
        require(!sender.voted, "you has voted"); // 确保发送者尚未投票
        require(to != msg.sender, "you can't do this operation"); // 确保不能委托给自己
        
        // 循环查找最终的委托投票地址
        while (voters[to].delegate != address(0) && voters[to].delegate != msg.sender)
            to = voters[to].delegate;

        sender.delegate = to; // 设置委托地址
        sender.voted = true;  // 标记发送者已投票
        
        Voter storage delegate_voter = voters[to]; // 获取被委托的投票者信息
        if (delegate_voter.voted == true) // 如果被委托的投票者已投票
            proposals[delegate_voter.vote] += sender.weight; // 将发送者的权重加到对应的提案上
        else
            delegate_voter.weight += sender.weight; // 否则，增加被委托者的权重
    }

    // 投票函数，接收提案索引
    function vote(uint8 proposal) public {
        Voter storage voter = voters[msg.sender]; // 获取当前投票者信息
        require(voter.voted == false, "you has voted"); // 确保投票者尚未投票
        voter.vote = proposal; // 设置投票者选择的提案
        voter.voted = true; // 标记投票者已投票
        proposals[proposal] += voter.weight; // 将投票者的权重加到相应提案的投票数上
    }

    // 返回获胜的提案索引
    function winningproposal() public view returns(uint8 winningone) {
        uint winningCount = 0; // 记录最高投票数
        for (uint8 i = 0; i < number_proposals; i++) {
            if (proposals[i] > winningCount) {
                winningCount = proposals[i]; // 更新最高投票数
                winningone = i; // 更新获胜提案索引
            }
        }
        return winningone; // 返回获胜提案的索引
    }

    // 检查特定投票者是否已投票
    function get(address vote_1) public view returns(bool) {
        require(msg.sender == chairperson, "you don't have the right"); // 只有主持人可以执行此操作
        return (voters[vote_1].voted); // 返回指定投票者的投票状态
    }
}

```

2. 部署脚本

```solidity
from brownie import *

ballot = Ballot.deploy(3,{'from':accounts[0]})

def giveRightToVote_deploy(id):
    ballot.giveRightToVote(accounts[id],{'from':accounts[0]})

def delegate_deploy(id_from,id_to):
    ballot.delegate(accounts[id_to], {'from': accounts[id_from]})

def vote_deploy(voter_id,proposal_id):
    ballot.vote(proposal_id,{'from':accounts[voter_id]})

def winningproposal_deploy(voter_id):
    win = ballot.winningproposal({'from':accounts[voter_id]})
    print(win)

def get_deploy(from_id,voter_id):
    voter_infor = ballot.get(accounts[voter_id],{'from':accounts[from_id]})
    print(voter_infor)

def main():
    for i in range(1,6):
        giveRightToVote_deploy(i)

    delegate_deploy(2,3)
    delegate_deploy(3,4)

    get_deploy(0,2)

    vote_deploy(1,1)
    vote_deploy(4,2)
    vote_deploy(5,0)

    winningproposal_deploy(0)
```
